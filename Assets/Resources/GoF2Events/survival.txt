# Survival: hold out for 3 minutes while pirates keep coming; new ones whenever fewer than 3 per player are left.
# Run it with /event survival.

set length = 180
score time   # the winner: alive the longest
set started = time

title @a Survival | Hold out for {length / 60} minutes for 5
wait 5
timer @a {length} Survive

while time - started < length + 5
  if inspace == 0
    timer @a stop
    title @a Defeat | Nobody made it for 5
    wait 5
    winner
    stop
  end
  if enemies < inspace * 3
    spawn @r Hiro {random(2, 4)} pirate enemy
    wait 3
  end
  wait 1
end

timer @a stop
title @a You survived! | {length / 60} minutes of pirates for 5
heal @a
reward @survivors 20000 | Survived
wait 5
winner
