# Waves: more pirates every wave; the next wave comes 5 s after the last one is destroyed.
# Run it with /event waves (admins) or "event waves" on the dedicated server's console.
# Copy it into an Events folder (see /event list) under another name to make your own.

set waves = 5
score kills   # the winner: the most pirates destroyed

title @a Waves | Survive {waves} waves of pirates for 5
timer @a 15 First wave
wait 15

repeat waves as wave
  title @a Wave {wave}/{waves} | Incoming for 4
  # Two more ships per wave ahead of each player in space (10 at most per player); a Bloodstar every third wave.
  spawn @a Hiro {min(wave * 2, 10)} pirate enemy
  if wave % 3 == 0
    spawn @a Bloodstar pirate enemy
  end
  wait until enemies == 0 or inspace == 0 timeout 600
  if inspace == 0
    title @a Defeat | Everyone was destroyed in wave {wave} for 5
    wait 5
    winner
    stop
  end
  if wave < waves
    title @a Wave {wave} cleared | Next wave in 5 seconds for 4
    # Everyone still alive since the fight began gets paid for the wave.
    reward @survivors {wave * 1000} | Wave {wave} survived
    heal @a
    ammo @a
    wait 5
  end
end

title @a Victory! | All {waves} waves cleared for 5
reward @survivors {waves * 5000} + Energy Cells 10 | All waves survived
wait 5
winner
